cmake_minimum_required(VERSION 3.22.1)
project(chroma_analysis LANGUAGES C CXX)

get_filename_component(LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../.model-cache/llama.cpp" ABSOLUTE)
if(NOT EXISTS "${LLAMA_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 scripts/prepare-analysis-android.py before building Android.")
endif()
get_filename_component(VULKAN_TOOLS "${CMAKE_CURRENT_LIST_DIR}/../../../../.model-cache/android-vulkan" ABSOLUTE)
if(NOT EXISTS "${VULKAN_TOOLS}/toolchain.cmake")
  message(FATAL_ERROR "Run python3 scripts/prepare-analysis-android.py to prepare Vulkan headers and host glslc.")
endif()
include("${VULKAN_TOOLS}/toolchain.cmake")
find_package(Vulkan COMPONENTS glslc REQUIRED)
# The shader generator runs on the build host. Android Gradle supplies Ninja by
# absolute path; it is not necessarily on PATH in the nested host CMake process.
find_program(CHROMA_HOST_CC NAMES cc gcc clang PATHS ENV PATH NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
find_program(CHROMA_HOST_CXX NAMES c++ g++ clang++ PATHS ENV PATH NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${CMAKE_CURRENT_BINARY_DIR}/chroma-vulkan-host.cmake" CACHE FILEPATH "" FORCE)
file(WRITE "${GGML_VULKAN_SHADERS_GEN_TOOLCHAIN}"
  "set(CMAKE_C_COMPILER [[${CHROMA_HOST_CC}]])\n"
  "set(CMAKE_CXX_COMPILER [[${CHROMA_HOST_CXX}]])\n"
  "set(CMAKE_MAKE_PROGRAM [[${CMAKE_MAKE_PROGRAM}]] CACHE FILEPATH \"\" FORCE)\n")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
target_include_directories(ggml-vulkan PRIVATE "${CHROMA_SPIRV_INCLUDE}")
add_library(chroma-analysis SHARED analysis-jni.cpp)
target_compile_features(chroma-analysis PRIVATE cxx_std_17)
target_link_libraries(chroma-analysis PRIVATE mtmd llama Vulkan::Vulkan android log)
target_link_options(chroma-analysis PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--wrap=vkGetPhysicalDeviceFeatures2")
